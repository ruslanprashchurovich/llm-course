def normalize_phone(text: str) -> str:
    # Удаляем все символы, кроме цифр
    cleaned_digits = ''.join(char for char in text if char.isdigit())
    
    # Проверка на пустую строку или отсутствие цифр
    if not cleaned_digits:
        raise ValueError("Некорректный номер телефона")

    # Преобразование префикса 8 в 7
    if cleaned_digits.startswith('8'):
        cleaned_digits = '7' + cleaned_digits[1:]
    
    # Проверка на корректную длину (ровно 11 цифр) и начало с 7
    if len(cleaned_digits) != 11 or not cleaned_digits.startswith('7'):
        raise ValueError("Некорректный номер телефона")

    return cleaned_digits
